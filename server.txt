/**
 * NEVIX Chat — сервер.
 * Что делает: аккаунты (хеш пароля scrypt), токены, чаты, сообщения,
 * непрочитанные и серверный подсчёт серий NEXUS STREAK.
 * Нужен только пакет pg (Postgres). Всё остальное — стандартная библиотека Node.
 */
const http = require('http');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const SESSION_DAYS = 90;
const AWARDS = [
  { at: 50, key: 'copper' },
  { at: 100, key: 'gold' },
  { at: 150, key: 'violet' },
  { at: 200, key: 'neon' }
];

/* ─────────── пароли и токены ─────────── */
function hashPassword(password) {
  const salt = crypto.randomBytes(16);
  const hash = crypto.scryptSync(password, salt, 64, { N: 16384, r: 8, p: 1 });
  return 'scrypt$' + salt.toString('hex') + '$' + hash.toString('hex');
}

function checkPassword(password, stored) {
  try {
    const parts = String(stored).split('$');
    if (parts.length !== 3) return false;
    const salt = Buffer.from(parts[1], 'hex');
    const hash = Buffer.from(parts[2], 'hex');
    const test = crypto.scryptSync(password, salt, hash.length);
    return crypto.timingSafeEqual(hash, test);
  } catch (e) {
    return false;
  }
}

function newToken() {
  return crypto.randomBytes(24).toString('hex');
}

function newId(prefix) {
  return prefix + '_' + crypto.randomBytes(8).toString('hex');
}

// «Сегодня» по часовому поясу сервера серий (по умолчанию Москва, +3)
const TZ_OFFSET = Number(process.env.STREAK_TZ_OFFSET || 3);
function today() {
  const now = new Date(Date.now() + TZ_OFFSET * 3600 * 1000);
  return now.toISOString().slice(0, 10);
}

function dayDiff(a, b) {
  return Math.round((Date.parse(a + 'T00:00:00Z') - Date.parse(b + 'T00:00:00Z')) / 86400000);
}

/* ─────────── схема (создаётся сама при старте) ─────────── */
const SCHEMA = [
  `create table if not exists users (
     id text primary key,
     lname text not null,
     name text not null,
     pass text not null,
     color text not null default '#ff3b3b',
     created timestamptz not null default now()
   )`,
  `create table if not exists tokens (
     token text primary key,
     user_id text not null,
     created timestamptz not null default now()
   )`,
  `create table if not exists chats (
     id text primary key,
     kind text not null,
     a text not null,
     b text,
     contact text,
     created timestamptz not null default now()
   )`,
  `create table if not exists messages (
     id text primary key,
     chat text not null,
     sender text not null,
     body text not null,
     seen boolean not null default false,
     created timestamptz not null default now()
   )`,
  `create table if not exists streaks (
     chat text primary key,
     count integer not null default 0,
     record integer not null default 0,
     counted text,
     day text,
     mine boolean not null default false,
     theirs boolean not null default false,
     awards text not null default ''
   )`,
  `create table if not exists reads (
     chat text not null,
     user_id text not null,
     seen_at timestamptz not null default now(),
     primary key (chat, user_id)
   )`
];

async function ensureSchema(pool) {
  for (const sql of SCHEMA) await pool.query(sql);
}

function botAnswer() {

}

/* ─────────── серии ─────────── */
async function getStreak(pool, chatId) {
  const r = await pool.query('select * from streaks where chat = $1', [chatId]);
  if (r.rows.length) return r.rows[0];
  await pool.query('insert into streaks (chat) values ($1)', [chatId]);
  return { chat: chatId, count: 0, record: 0, counted: null, day: null, mine: false, theirs: false, awards: '' };
}

// who: 'mine' — написал владелец чата, 'theirs' — собеседник
async function markStreak(pool, chatId, who) {
  const st = await getStreak(pool, chatId);
  const t = today();
  let count = Number(st.count) || 0;
  let mine = !!st.mine;
  let theirs = !!st.theirs;
  let day = st.day;
  let counted = st.counted;
  let record = Number(st.record) || 0;
  let awards = st.awards || '';

  if (day !== t) {
    if (day && dayDiff(t, day) > 1) count = 0; // пропустили день — серия обрывается
    mine = false;
    theirs = false;
    day = t;
  }

  if (who === 'mine') mine = true;
  else theirs = true;

  let fresh = false;
  if (mine && theirs && counted !== t) {
    count += 1;
    counted = t;
    fresh = true;
    if (count > record) record = count;
    const got = awards ? awards.split(',') : [];
    AWARDS.forEach((x) => { if (count >= x.at && got.indexOf(x.key) === -1) got.push(x.key); });
    awards = got.join(',');
  }

  await pool.query(
    'update streaks set count = $2, record = $3, counted = $4, day = $5, mine = $6, theirs = $7, awards = $8 where chat = $1',
    [chatId, count, record, counted, day, mine, theirs, awards]
  );

  return { count, record, awards: awards ? awards.split(',') : [], mine, theirs, day, fresh };
}

/* ─────────── разбор запросов ─────────── */
function json(res, code, data) {
  const body = JSON.stringify(data);
  res.writeHead(code, {
    'content-type': 'application/json; charset=utf-8',
    'cache-control': 'no-store',
    'access-control-allow-origin': '*',
    'access-control-allow-headers': 'content-type, authorization',
    'access-control-allow-methods': 'GET, POST, OPTIONS'
  });
  res.end(body);
}

function readBody(req) {
  return new Promise((resolve, reject) => {
    let size = 0;
    const chunks = [];
    req.on('data', (c) => {
      size += c.length;
      if (size > 200 * 1024) { reject(new Error('слишком большое тело запроса')); req.destroy(); return; }
      chunks.push(c);
    });
    req.on('end', () => {
      if (!chunks.length) return resolve({});
      try { resolve(JSON.parse(Buffer.concat(chunks).toString('utf8'))); }
      catch (e) { reject(new Error('не разобрал JSON')); }
    });
    req.on('error', reject);
  });
}

async function auth(pool, req) {
  const h = req.headers.authorization || '';
  const token = h.startsWith('Bearer ') ? h.slice(7).trim() : '';
  if (!token) return null;
  const r = await pool.query('select * from tokens where token = $1', [token]);
  if (!r.rows.length) return null;
  const u = await pool.query('select * from users where id = $1', [r.rows[0].user_id]);
  if (!u.rows.length) return null;
  return u.rows[0];
}

function userJson(u) {
  return { id: u.id, name: u.name, color: u.color };
}

/* ─────────── чат в вид, удобный приложению ─────────── */
async function chatJson(pool, chat, me) {
  const st = await getStreak(pool, chat.id);
  const lastR = await pool.query(
    'select * from messages where chat = $1 order by created desc, id desc limit 1',
    [chat.id]
  );
  const readR = await pool.query('select seen_at from reads where chat = $1 and user_id = $2', [chat.id, me.id]);
  const readAt = readR.rows.length ? readR.rows[0].seen_at : null;
  const unreadR = readAt
    ? await pool.query('select count(*) as c from messages where chat = $1 and sender <> $2 and created > $3', [chat.id, me.id, readAt])
    : await pool.query('select count(*) as c from messages where chat = $1 and sender <> $2', [chat.id, me.id]);

  let title = 'Чат';
  let color = '#3f7d5a';
  let online = false;
  let about = 'был(а) недавно';
  let peerId = null;

  if (chat.kind === 'bot') {
    const bot = BOTS.find((b) => b.id === chat.contact) || BOTS[0];
    title = bot.name;
    color = bot.color;
    online = bot.online;
    about = bot.about;
  } else {
    peerId = chat.a === me.id ? chat.b : chat.a;
    const p = await pool.query('select * from users where id = $1', [peerId]);
    if (p.rows.length) {
      title = p.rows[0].name;
      color = p.rows[0].color;
      online = true;
      about = 'в сети';
    }
  }

  const last = lastR.rows.length
    ? { text: lastR.rows[0].body, me: lastR.rows[0].sender === me.id, time: lastR.rows[0].created }
    : null;

  return {
    id: chat.id,
    kind: chat.kind,
    title,
    color,
    online,
    about,
    peerId,
    unread: Number(unreadR.rows[0].c) || 0,
    last,
    streak: {
      count: Number(st.count) || 0,
      record: Number(st.record) || 0,
      awards: st.awards ? st.awards.split(',') : [],
      mine: !!st.mine,
      theirs: !!st.theirs,
      day: st.day
    }
  };
}

async function chatById(pool, id) {
  const r = await pool.query('select * from chats where id = $1', [id]);
  return r.rows[0] || null;
}

function member(chat, userId) {
  return chat.a === userId || chat.b === userId;
}

/* ─────────── приложение ─────────── */
function createServer(options) {
  const pool = options.pool;
  const publicDir = options.publicDir || __dirname;

  const server = http.createServer(async (req, res) => {
    const url = new URL(req.url, 'http://localhost');
    const p = url.pathname.replace(/\/+$/, '') || '/';

    if (req.method === 'OPTIONS') return json(res, 204, {});

    try {
      /* --- служебное --- */
      if (p === '/api/health') return json(res, 200, { ok: true, time: new Date().toISOString() });

      /* --- регистрация и вход --- */
      if (p === '/api/auth/signup' && req.method === 'POST') {
        const body = await readBody(req);
        const name = String(body.name || '').trim();
        const password = String(body.password || '');
        if (name.length < 2) return json(res, 400, { error: 'Имя — минимум 2 символа' });
        if (name.length > 24) return json(res, 400, { error: 'Имя — максимум 24 символа' });
        if (password.length < 6) return json(res, 400, { error: 'Пароль — минимум 6 символов' });

        const lname = name.toLowerCase();
        const busy = await pool.query('select id from users where lname = $1', [lname]);
        if (busy.rows.length) return json(res, 409, { error: 'Такое имя уже занято' });

        const colors = ['#3f7d5a', '#8a5b9e', '#2f6f9e', '#b4762f', '#9e4b4b', '#ff3b3b'];
        const user = {
          id: newId('u'),
          lname,
          name,
          pass: hashPassword(password),
          color: colors[Math.floor(Math.random() * colors.length)]
        };
        await pool.query('insert into users (id, lname, name, pass, color) values ($1,$2,$3,$4,$5)',
          [user.id, user.lname, user.name, user.pass, user.color]);

        const token = newToken();
        await pool.query('insert into tokens (token, user_id) values ($1,$2)', [token, user.id]);
        return json(res, 201, { token, user: userJson(user) });
      }

      if (p === '/api/auth/login' && req.method === 'POST') {
        const body = await readBody(req);
        const name = String(body.name || '').trim().toLowerCase();
        const password = String(body.password || '');
        const r = await pool.query('select * from users where lname = $1', [name]);
        if (!r.rows.length || !checkPassword(password, r.rows[0].pass)) {
          return json(res, 401, { error: 'Неверное имя или пароль' });
        }
        const token = newToken();
        await pool.query('insert into tokens (token, user_id) values ($1,$2)', [token, r.rows[0].id]);
        return json(res, 200, { token, user: userJson(r.rows[0]) });
      }

      if (p === '/api/auth/logout' && req.method === 'POST') {
        const h = req.headers.authorization || '';
        const token = h.startsWith('Bearer ') ? h.slice(7).trim() : '';
        if (token) await pool.query('delete from tokens where token = $1', [token]);
        return json(res, 200, { ok: true });
      }

      /* --- дальше всё требует токен --- */
      const me = await auth(pool, req);

      if (p === '/api/me' && req.method === 'GET') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        return json(res, 200, { user: userJson(me) });
      }

      /* сменить имя и цвет */
      if (p === '/api/me' && req.method === 'POST') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const body = await readBody(req);
        const name = String(body.name || me.name).trim();
        if (name.length < 2) return json(res, 400, { error: 'Имя — минимум 2 символа' });
        const color = String(body.color || me.color);
        const lname = name.toLowerCase();
        const busy = await pool.query('select id from users where lname = $1 and id <> $2', [lname, me.id]);
        if (busy.rows.length) return json(res, 409, { error: 'Такое имя уже занято' });
        await pool.query('update users set name = $2, lname = $3, color = $4 where id = $1', [me.id, name, lname, color]);
        return json(res, 200, { user: { id: me.id, name, color } });
      }

      /* смена пароля */
      if (p === '/api/auth/password' && req.method === 'POST') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const body = await readBody(req);
        const oldPass = String(body.oldPassword || '');
        const newPass = String(body.newPassword || '');
        if (newPass.length < 6) return json(res, 400, { error: 'Новый пароль — минимум 6 символов' });
        if (!checkPassword(oldPass, me.pass)) return json(res, 400, { error: 'Текущий пароль неверный' });
        await pool.query('update users set pass = $2 where id = $1', [me.id, hashPassword(newPass)]);
        return json(res, 200, { ok: true });
      }

      /* удалить аккаунт со всеми чатами */
      if (p === '/api/auth/delete' && req.method === 'POST') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const mine = await pool.query('select * from chats where a = $1 or b = $1', [me.id]);
        for (const c of mine.rows) {
          await pool.query('delete from messages where chat = $1', [c.id]);
          await pool.query('delete from streaks where chat = $1', [c.id]);
          await pool.query('delete from reads where chat = $1', [c.id]);
          await pool.query('delete from chats where id = $1', [c.id]);
        }
        await pool.query('delete from tokens where user_id = $1', [me.id]);
        await pool.query('delete from users where id = $1', [me.id]);
        return json(res, 200, { ok: true });
      }

      if (p === '/api/contacts' && req.method === 'GET') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const r = await pool.query('select id, name, color from users where id <> $1 order by name', [me.id]);
        return json(res, 200, { bots: BOTS, users: r.rows });
      }

      if (p === '/api/chats' && req.method === 'GET') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const r = await pool.query('select * from chats where a = $1 or b = $1 order by created asc', [me.id]);
        const out = [];
        for (const c of r.rows) out.push(await chatJson(pool, c, me));
        return json(res, 200, { chats: out });
      }

      /* создать чат: с живым человеком по имени или с демо-собеседником */
      if (p === '/api/chats' && req.method === 'POST') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const body = await readBody(req);

        if (body.contactId) {
          const bot = BOTS.find((b) => b.id === body.contactId);
          if (!bot) return json(res, 404, { error: 'Нет такого собеседника' });
          const id = 'bot:' + me.id + ':' + bot.id;
          let chat = await chatById(pool, id);
          if (!chat) {
            await pool.query('insert into chats (id, kind, a, contact) values ($1,$2,$3,$4)',
              [id, 'bot', me.id, bot.id]);
            chat = await chatById(pool, id);
          }
          return json(res, 200, { chat: await chatJson(pool, chat, me) });
        }

        const peerName = String(body.peerName || '').trim().toLowerCase();
        if (peerName.length < 2) return json(res, 400, { error: 'Введите имя человека' });
        const peerR = await pool.query('select * from users where lname = $1', [peerName]);
        if (!peerR.rows.length) return json(res, 404, { error: 'Человек с таким именем не зарегистрирован' });
        const peer = peerR.rows[0];
        if (peer.id === me.id) return json(res, 400, { error: 'Это вы' });

        const pair = [me.id, peer.id].sort();
        const id = 'd:' + pair[0] + ':' + pair[1];
        let chat = await chatById(pool, id);
        if (!chat) {
          await pool.query('insert into chats (id, kind, a, b) values ($1,$2,$3,$4)',
            [id, 'direct', pair[0], pair[1]]);
          chat = await chatById(pool, id);
        }
        return json(res, 200, { chat: await chatJson(pool, chat, me) });
      }

      /* сообщения чата */
      const msgMatch = p.match(/^\/api\/chats\/([^/]+)\/messages$/);
      if (msgMatch) {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const chatId = decodeURIComponent(msgMatch[1]);
        const chat = await chatById(pool, chatId);
        if (!chat || !member(chat, me.id)) return json(res, 404, { error: 'Нет такого чата' });

        if (req.method === 'GET') {
          const since = url.searchParams.get('since');
          const r = since
            ? await pool.query('select * from messages where chat = $1 and created > $2 order by created asc', [chatId, since])
            : await pool.query('select * from messages where chat = $1 order by created asc', [chatId]);
          return json(res, 200, {
            messages: r.rows.map((m) => ({
              id: m.id,
              text: m.body,
              me: m.sender === me.id,
              time: m.created,
              seen: !!m.seen
            }))
          });
        }

        if (req.method === 'POST') {
          const body = await readBody(req);
          const text = String(body.text || '').trim();
          if (!text) return json(res, 400, { error: 'Пустое сообщение' });
          if (text.length > 4000) return json(res, 400, { error: 'Слишком длинное сообщение' });

          const id = newId('m');
          await pool.query('insert into messages (id, chat, sender, body) values ($1,$2,$3,$4)',
            [id, chatId, me.id, text]);
          // в личном чате «своя» сторона — та, что записана в chat.a
          const side = chat.kind === 'direct' ? (me.id === chat.a ? 'mine' : 'theirs') : 'mine';
          const streak = await markStreak(pool, chatId, side);

          if (chat.kind === 'bot') {
            const peer = BOTS.find((b) => b.id === chat.contact) || BOTS[0];
            setTimeout(async () => {
              try {
                const answer = botAnswer();
                await pool.query('insert into messages (id, chat, sender, body) values ($1,$2,$3,$4)',
                  [newId('m'), chatId, 'bot:' + peer.id, answer]);
                await markStreak(pool, chatId, 'theirs');
              } catch (e) { /* сервер мог уснуть — это нормально для демо */ }
            }, 1200);
          }

          const r = await pool.query('select * from messages where id = $1', [id]);
          return json(res, 201, {
            message: { id, text, me: true, time: r.rows[0].created, seen: false },
            streak
          });
        }
      }

      /* отметка «прочитано» */
      const readMatch = p.match(/^\/api\/chats\/([^/]+)\/read$/);
      if (readMatch && req.method === 'POST') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const chatId = decodeURIComponent(readMatch[1]);
        const chat = await chatById(pool, chatId);
        if (!chat || !member(chat, me.id)) return json(res, 404, { error: 'Нет такого чата' });
        const now = new Date().toISOString();
        await pool.query('delete from reads where chat = $1 and user_id = $2', [chatId, me.id]);
        await pool.query('insert into reads (chat, user_id, seen_at) values ($1,$2,$3)', [chatId, me.id, now]);
        return json(res, 200, { ok: true });
      }

      /* серия */
      const stMatch = p.match(/^\/api\/chats\/([^/]+)\/streak$/);
      if (stMatch && req.method === 'GET') {
        if (!me) return json(res, 401, { error: 'Нужен вход' });
        const chatId = decodeURIComponent(stMatch[1]);
        const chat = await chatById(pool, chatId);
        if (!chat || !member(chat, me.id)) return json(res, 404, { error: 'Нет такого чата' });
        const st = await getStreak(pool, chatId);
        return json(res, 200, {
          streak: {
            count: Number(st.count) || 0,
            record: Number(st.record) || 0,
            awards: st.awards ? st.awards.split(',') : []
          }
        });
      }

      if (p.startsWith('/api/')) return json(res, 404, { error: 'Нет такого адреса' });

      /* --- статика: само приложение --- */
      let file = path.join(publicDir, p === '/' ? 'index.html' : p.replace(/^\/+/, ''));
      if (!file.startsWith(publicDir)) file = path.join(publicDir, 'index.html');
      fs.readFile(file, (err, data) => {
        if (err) {
          fs.readFile(path.join(publicDir, 'index.html'), (e2, html) => {
            if (e2) { res.writeHead(404); return res.end('Не найдено'); }
            res.writeHead(200, { 'content-type': 'text/html; charset=utf-8' });
            res.end(html);
          });
          return;
        }
        const ext = path.extname(file);
        const types = { '.html': 'text/html; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.css': 'text/css; charset=utf-8', '.ico': 'image/x-icon', '.png': 'image/png', '.svg': 'image/svg+xml' };
        res.writeHead(200, { 'content-type': types[ext] || 'application/octet-stream' });
        res.end(data);
      });
    } catch (e) {
      json(res, 500, { error: 'Ошибка сервера: ' + e.message });
    }
  });

  server.ensureSchema = () => ensureSchema(pool);
  return server;
}

/* ─────────── запуск ─────────── */
if (require.main === module) {
  const { Pool } = require('pg');
  const connectionString = process.env.DATABASE_URL;
  if (!connectionString) {
    console.error('Нет переменной DATABASE_URL — не к чему подключаться.');
    process.exit(1);
  }
  const pool = new Pool({
    connectionString,
    ssl: connectionString.includes('sslmode=disable') ? false : { rejectUnauthorized: false },
    max: 5
  });

  const server = createServer({ pool });

  server.ensureSchema()
    .then(() => {
      const port = process.env.PORT || 3000;
      server.listen(port, '0.0.0.0', () => {
        console.log('NEVIX Chat сервер слушает порт ' + port);
      });
    })
    .catch((e) => {
      console.error('Не смог создать таблицы: ' + e.message);
      process.exit(1);
    });
}

module.exports = { createServer, ensureSchema, hashPassword, checkPassword, markStreak, today };
